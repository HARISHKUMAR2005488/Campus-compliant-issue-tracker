    async function handleStudentRegister() {
      const name = document.getElementById('reg-name').value.trim();
      const studentId = document.getElementById('reg-studentid').value.trim();
      const department = document.getElementById('reg-department').value;
      const email = document.getElementById('reg-email').value.trim();
      const password = document.getElementById('reg-password').value;

      if (!name || !studentId || !department || !email || !password) {
        showToast('Please fill in all fields', 'error');
        return;
      }

      if (password.length < 6) {
        showToast('Password must be at least 6 characters', 'error');
        return;
      }

      const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
      if (!emailRegex.test(email)) {
        showToast('Please enter a valid email', 'error');
        return;
      }

      if (allData.length >= 999) {
        showToast('Maximum data limit reached', 'error');
        return;
      }

      const existingUser = allData.find(d => d.type === 'user' && d.email === email);
      if (existingUser) {
        showToast('Email already registered', 'error');
        return;
      }

      showLoading(true);

      const result = await window.dataSdk.create({
        type: 'user',
        name,
        studentId,
        department,
        email,
        password,
        createdAt: new Date().toISOString()
      });

      if (result.isOk) {
        showToast('Account created! Please login', 'success');
        switchAuthTab('login');
        document.getElementById('login-email').value = email;
        document.getElementById('reg-name').value = '';
        document.getElementById('reg-studentid').value = '';
        document.getElementById('reg-department').value = '';
        document.getElementById('reg-email').value = '';
        document.getElementById('reg-password').value = '';
      } else {
        showToast('Registration failed. Try again.', 'error');
      }

      showLoading(false);
    }

    // ========== Statistics ==========
    function updateAllStats() {
      const issues = allData.filter(d => d.type === 'issue');
      
      // Student stats (filtered by current user)
      if (currentUser && currentUser.type !== 'admin') {
        const userIssues = issues.filter(i => i.reportedBy === currentUser.email);
        document.getElementById('stat-student-total').textContent = userIssues.length;
        document.getElementById('stat-student-pending').textContent = userIssues.filter(i => i.status === 'Pending').length;
        document.getElementById('stat-student-progress').textContent = userIssues.filter(i => i.status === 'In Progress').length;
        document.getElementById('stat-student-resolved').textContent = userIssues.filter(i => i.status === 'Resolved').length;
      }

      // Admin stats (all issues)
      document.getElementById('stat-admin-total').textContent = issues.length;
      document.getElementById('stat-admin-pending').textContent = issues.filter(i => i.status === 'Pending').length;
      document.getElementById('stat-admin-progress').textContent = issues.filter(i => i.status === 'In Progress').length;
      document.getElementById('stat-admin-resolved').textContent = issues.filter(i => i.status === 'Resolved').length;

      // Calculate average resolution time
      const resolved = issues.filter(i => i.status === 'Resolved' && i.resolvedAt && i.createdAt);
      if (resolved.length > 0) {
        let totalHours = 0;
        resolved.forEach(i => {
          const created = new Date(i.createdAt);
          const resolvedAt = new Date(i.resolvedAt);
          totalHours += (resolvedAt - created) / (1000 * 60 * 60);
        });
        const avg = (totalHours / resolved.length).toFixed(1);
        document.getElementById('avg-resolution-time').textContent = avg + ' hrs';
      } else {
        document.getElementById('avg-resolution-time').textContent = '-- hrs';
      }
    }

    function setAdminStatus(status) {
      if (!currentIssue) return;
      currentIssue.status = status;
      
      document.querySelectorAll('.admin-status-btn').forEach(btn => {
        btn.classList.remove('border-yellow-500', 'bg-yellow-50', 'text-yellow-700');
        btn.classList.remove('border-blue-500', 'bg-blue-50', 'text-blue-700');
        btn.classList.remove('border-green-500', 'bg-green-50', 'text-green-700');
        btn.classList.add('border-gray-200', 'text-gray-600');
      });

      const selected = document.querySelector(`[data-status="${status}"]`);
      if (selected) {
        selected.classList.remove('border-gray-200', 'text-gray-600');
        if (status === 'Pending') {
          selected.classList.add('border-yellow-500', 'bg-yellow-50', 'text-yellow-700');
        } else if (status === 'In Progress') {
          selected.classList.add('border-blue-500', 'bg-blue-50', 'text-blue-700');
        } else {
          selected.classList.add('border-green-500', 'bg-green-50', 'text-green-700');
        }
      }
    }

    function setAdminPriority(priority) {
      if (!currentIssue) return;
      currentIssue.priority = priority;
      
      document.querySelectorAll('.admin-priority-btn').forEach(btn => {
        btn.classList.remove('border-indigo-500', 'bg-indigo-50', 'text-indigo-700');
        btn.classList.remove('border-yellow-500', 'bg-yellow-50', 'text-yellow-700');
        btn.classList.remove('border-red-500', 'bg-red-50', 'text-red-700');
        btn.classList.add('border-gray-200', 'text-gray-600');
      });

      const selected = document.querySelector(`[data-priority="${priority}"]`);
      if (selected) {
        selected.classList.remove('border-gray-200', 'text-gray-600');
        if (priority === 'Low') {
          selected.classList.add('border-indigo-500', 'bg-indigo-50', 'text-indigo-700');
        } else if (priority === 'Medium') {
          selected.classList.add('border-yellow-500', 'bg-yellow-50', 'text-yellow-700');
        } else {
          selected.classList.add('border-red-500', 'bg-red-50', 'text-red-700');
        }
      }
    }

    async function saveAdminChanges() {
      if (!currentIssue) return;

      showLoading(true);

      const comment = document.getElementById('admin-comment')?.value || '';
      const updatedIssue = {
        ...currentIssue,
        adminComment: comment,
        updatedAt: new Date().toISOString(),
        resolvedAt: currentIssue.status === 'Resolved' && !currentIssue.resolvedAt 
          ? new Date().toISOString() 
          : currentIssue.resolvedAt
      };

      const result = await window.dataSdk.update(updatedIssue);

      if (result.isOk) {
        showToast('Issue updated successfully!', 'success');
        showScreen('admin-dashboard');
      } else {
        showToast('Failed to update issue', 'error');
      }

      showLoading(false);
    }

    function formatDate(isoString) {
      if (!isoString) return '';
      const date = new Date(isoString);
      return date.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
    }

    function formatDateTime(isoString) {
      if (!isoString) return '';
      const date = new Date(isoString);
      return date.toLocaleDateString('en-US', { 
        month: 'short', 
        day: 'numeric', 
        year: 'numeric',
        hour: '2-digit',
        minute: '2-digit'
      });
    }
